BIT

一点加算と区間和を $O(\log N)$ で行います.インデックスは 1-indexed です.
参照URL：https://salt-lettuce.github.io/algo_notes_by_lettuce/library/#bit

# https://salt-lettuce.github.io/algo_notes_by_lettuce/library/#bit
class BIT:
    # 長さN+1の配列を初期化
    def __init__(self, N):
        self.size = N
        self.bit = [0]*(N+1)

    # i番目までの和を求める
    def sum(self, i):
        res = 0
        while i > 0: # bit[0]=0なので計算しない
            res += self.bit[i] # フェニック木のi番目の値を加算
            i -= -i & i # 最も右にある1の桁を0にする
        return res

    # i番目の値にxを足して更新する
    def add(self, i, x):
        while i <= self.size: # iがNを超えないように制限
            self.bit[i] += x # フェニック木のi番目にxを足して更新
            i += -i & i # 最も右にある1の桁に1を足す

BinarySearch

単調増加する関数 $f(x)$ について,「$f(x) \ge$ target となる最小の $x$」「$f(x) \le$ target となる最大の $x$」を求めます.
参照URL：https://salt-lettuce.github.io/algo_notes_by_lettuce/library/#binary-search

# https://salt-lettuce.github.io/algo_notes_by_lettuce/library/#binary-search
class BinarySearch:
    # f は単調増加(x が大きいほど f(x) も大きいか等しい)な関数
    def __init__(self, f):
        self.f = f

    # f(x) >= target となる最小の整数 x を [lo, hi] から探す (なければ hi + 1)
    def min_x(self, target, lo, hi):
        ng, ok = lo - 1, hi + 1 # ng 側は常に条件を満たさない,ok 側は常に満たすとみなす
        while ok - ng > 1:
            mid = (ok + ng) // 2
            if self.f(mid) >= target:
                ok = mid # 答えは mid 以下
            else:
                ng = mid # 答えは mid より大きい
        return ok

    # f(x) <= target となる最大の整数 x を [lo, hi] から探す (なければ lo - 1)
    def max_x(self, target, lo, hi):
        ok, ng = lo - 1, hi + 1
        while ng - ok > 1:
            mid = (ok + ng) // 2
            if self.f(mid) <= target:
                ok = mid # 答えは mid 以上
            else:
                ng = mid # 答えは mid より小さい
        return ok

    # f(x) >= target となる最小の実数 x を [lo, hi] から探す (回数を決めて範囲を半分にし続ける)
    def min_x_real(self, target, lo, hi, n_iter=100):
        for _ in range(n_iter):
            mid = (lo + hi) / 2
            if self.f(mid) >= target:
                hi = mid
            else:
                lo = mid
        return hi
